package management

import (
	"context"
	"sync/atomic"
	"testing"
	"time"

	"github.com/masgzy/CloudflareST/internal/integrations"
)

func TestConfirmationConcurrentTargetWritesKeepActualOrder(test *testing.T) {
	started := make(chan string, 2)
	release := make(chan struct{}, 2)
	var active, overlap atomic.Int32
	provider := newAcceptanceProvider()
	provider.update = func(ctx context.Context, _ integrations.Connection, _ integrations.Target, ip string) integrations.Outcome {
		if active.Add(1) > 1 {
			overlap.Add(1)
		}
		defer active.Add(-1)
		started <- ip
		select {
		case <-release:
		case <-ctx.Done():
		}
		return integrations.Outcome{State: "success", Verified: true}
	}
	fixture := newAcceptanceFixture(test, acceptanceNow(), nil, provider)
	test.Cleanup(func() { close(release) })
	acceptanceConnection(test, fixture, "serial-connection-a", "alidns", "serial-account")
	acceptanceConnection(test, fixture, "serial-connection-b", "alidns", "serial-account")
	targetA := acceptanceTarget("serial-connection-a", "serial-account", "serial-record")
	targetB := acceptanceTarget("serial-connection-b", "serial-account", "serial-record")
	runs := []Run{
		{ID: "serial-run-a", TaskID: "serial-task-a", Status: "completed", Sequence: 1, Snapshot: acceptanceTask("人工推送A"), Deliveries: []Delivery{{ID: "serial-delivery-a", Kind: "dns", ConnectionID: targetA.ConnectionID, Target: &targetA, IP: "192.0.2.10", State: "pending"}}},
		{ID: "serial-run-b", TaskID: "serial-task-b", Status: "completed", Sequence: 2, Snapshot: acceptanceTask("人工推送B"), Deliveries: []Delivery{{ID: "serial-delivery-b", Kind: "dns", ConnectionID: targetB.ConnectionID, Target: &targetB, IP: "192.0.2.20", State: "pending"}}},
	}
	fixture.service.mu.Lock()
	for _, run := range runs {
		if err := put(fixture.service.db, "runs", run.ID, run); err != nil {
			fixture.service.mu.Unlock()
			test.Fatal(err)
		}
	}
	err := fixture.service.dispatch()
	fixture.service.mu.Unlock()
	if err != nil {
		test.Fatal(err)
	}
	awaitWrite := func() string {
		test.Helper()
		select {
		case ip := <-started:
			return ip
		case <-time.After(3 * time.Second):
			test.Fatal("替身DNS请求未在限时内开始")
			return ""
		}
	}
	first := awaitWrite()
	select {
	case second := <-started:
		test.Fatalf("首个请求尚未结束时不应发起同目标写入：%s、%s", first, second)
	case <-time.After(80 * time.Millisecond):
	}
	release <- struct{}{}
	second := awaitWrite()
	release <- struct{}{}
	for _, run := range runs {
		acceptanceWaitDelivery(test, fixture, run.ID, run.Deliveries[0].ID, "success")
	}
	metadata := targetA
	fixture.service.mu.Lock()
	err = fixture.service.appendTargetMetadata(fixture.service.db, &metadata)
	fixture.service.mu.Unlock()
	if err != nil || metadata.LastKnownWrite == nil || metadata.LastKnownWrite.IP != second {
		test.Fatalf("最近写入必须记录实际后写入值：%+v，%v", metadata.LastKnownWrite, err)
	}
	expectedRun := "serial-run-a"
	if second == "192.0.2.20" {
		expectedRun = "serial-run-b"
	}
	updates, notifications := provider.callCounts()
	if overlap.Load() != 0 || updates != 2 || notifications != 0 || first == second || metadata.LastKnownWrite.RunID != expectedRun {
		test.Fatalf("同目标互斥或最近写入事实不正确：重叠=%d，DNS/通知=%d/%d，先后=%s/%s，最近=%+v", overlap.Load(), updates, notifications, first, second, metadata.LastKnownWrite)
	}
	test.Logf("同目标不同连接别名的人工投递：实际先后=%s→%s，重叠请求=0，最近写入IP=%s；未断言受理FIFO；全部为替身", first, second, metadata.LastKnownWrite.IP)
}
